### Обновление самоподписного сертификата

##### Как убрать сообщение Соединение $${\color{#FF0900}НЕ}$$ $${\color{#FF0900}ЗАЩИЩЕНО}$$ 

<img width="1550" height="483" alt="image" src="https://github.com/user-attachments/assets/53f7e0ad-ffa6-4685-bf5c-7750519253c1" />

Пишем скрипт для Cromium ``chromium_certs.sh``
```bash
#!/bin/bash

# Директория с сертификатами (можно изменить или передавать как аргумент)
SOURCE_DIR="${1:-/mnt/CDDiskPack/CDinstall_Astra_1.7.4/sslcert}"
# Пользователь, для которого обновляем сертификаты
TARGET_USER="cyber-root"
# База NSS пользователя
NSSDB_DIR="/home/$TARGET_USER/.pki/nssdb"

# Проверка наличия certutil
if ! command -v certutil &> /dev/null; then
    echo "Ошибка: утилита certutil не найдена. Установите пакет libnss3-tools." >&2
    exit 1
fi

# Проверка существования исходной директории
if [ ! -d "$SOURCE_DIR" ]; then
    echo "Ошибка: директория $SOURCE_DIR не существует." >&2
    exit 1
fi

# Функция для выполнения команд от имени целевого пользователя
run_as_user() {
    sudo -u "$TARGET_USER" "$@"
}

# Функция для импорта сертификата с удалением старого
import_cert() {
    local cert_name="$1"
    local cert_file="$2"
    local trust_args="$3"

    # Удаляем старый сертификат, если есть (игнорируем ошибку)
    run_as_user certutil -d "sql:$NSSDB_DIR" -D -n "$cert_name" 2>/dev/null || true

    # Импортируем новый
    if [ -f "$SOURCE_DIR/$cert_file" ]; then
        if run_as_user certutil -d "sql:$NSSDB_DIR" -A -t "$trust_args" -n "$cert_name" -i "$SOURCE_DIR/$cert_file"; then
            echo "  [+] Сертификат '$cert_name' успешно добавлен."
        else
            echo "  [X] Ошибка при добавлении сертификата '$cert_name'." >&2
        fi
    else
        echo "  [-] Файл '$cert_file' не найден, пропускаем импорт '$cert_name'." >&2
    fi
}

echo "=== Обновление сертификатов в базе NSS пользователя $TARGET_USER (Chromium) ==="
echo "Директория с сертификатами: $SOURCE_DIR"
echo

# Импорт корневого сертификата
import_cert "Russian Trusted Root" "Корн Russian Trusted root.pem" "C,,"

# Импорт промежуточного сертификата
import_cert "Russian Trusted Sub" "Корн Russian Trusted sub.pem" "C,,"

# Импорт сертификатов сайтов
import_cert "r16.rosreestr.ru" "r16.rosreestr.ru.crt" "P,,"
import_cert "rosreestr.ru" "rosreestr.ru.crt" "P,,"

echo "=== Готово ==="

# Для проверки можно вывести список установленных сертификатов:
# run_as_user certutil -d "sql:$NSSDB_DIR" -L
```

Пишем скрипт ``install_certs.sh`` для смены сертификатов в папках 
```bash
ls -la /ds-data/ssl/
ls -la /etc/nginx/ssl/
ls -la /var/www/r7-office/Data/ssl/
```
в пользовательской базе NSS (Chromium) и в профиле .r7organizer (NSS) Organizer
```bash
/home/cyber-root/.r7organizer/*.default-default# certutil -L -d sql:.
sudo -u cyber-root certutil -d sql:/home/cyber-root/.pki/nssdb -L
```
Скрипт install_certs.sh
```bash
#!/bin/bash
set -euo pipefail

# === Конфигурация ===
SOURCE_DIR="/mnt/CDDiskPack/CDinstall_Astra_1.7.4/sslcert"
TARGET_DIRS=(
    "/var/www/r7-office/Data/ssl"
    "/ds-data/ssl"
    "/etc/nginx/ssl"
)
PROFILE_PATTERN="/home/cyber-root/.r7organizer/*.default-default"
# Добавлено второе имя для обновления rosreestr.ru
CERT_NAMES=("r16.rosreestr.ru" "rosreestr.ru")

# === Проверка прав ===
if [ "$EUID" -ne 0 ]; then
    echo "Ошибка: скрипт должен быть запущен от root" >&2
    exit 1
fi

# === Проверка исходных файлов ===
missing=0
for name in "${CERT_NAMES[@]}"; do
    if [ ! -f "$SOURCE_DIR/${name}.crt" ]; then
        echo "Ошибка: исходный файл $SOURCE_DIR/${name}.crt не найден" >&2
        missing=1
    fi
    if [ ! -f "$SOURCE_DIR/${name}.key" ]; then
        echo "Предупреждение: исходный ключ $SOURCE_DIR/${name}.key не найден" >&2
    fi
done
if [ $missing -eq 1 ]; then
    exit 1
fi

# === Функция выполнения от имени cyber-root ===
run_as_cyber_root() {
    sudo -u cyber-root "$@"
}

# === 1. Копирование файлов в целевые папки ===
echo "=== Копирование сертификатов в целевые каталоги ==="
for dir in "${TARGET_DIRS[@]}"; do
    mkdir -p "$dir"
    echo "--> $dir"
    for name in "${CERT_NAMES[@]}"; do
        for ext in crt key; do
            src="$SOURCE_DIR/${name}.${ext}"
            if [ -f "$src" ]; then
                cp -v "$src" "$dir/"
                if [ "$ext" = "key" ]; then
                    chmod 600 "$dir/${name}.${ext}"
                fi
            fi
        done
    done
done

# === 2. Работа с профилем .r7organizer (NSS) ===
PROFILE_DIR=$(ls -d $PROFILE_PATTERN 2>/dev/null | head -n1)
if [ -z "$PROFILE_DIR" ]; then
    echo "ОШИБКА: Каталог профиля не найден по шаблону $PROFILE_PATTERN" >&2
else
    echo "=== Обновление сертификатов в профиле NSS: $PROFILE_DIR ==="
    # Копируем свежие файлы в профиль для всех имён
    for name in "${CERT_NAMES[@]}"; do
        for ext in crt key; do
            src="$SOURCE_DIR/${name}.${ext}"
            if [ -f "$src" ]; then
                cp -v "$src" "$PROFILE_DIR/"
            fi
        done
    done

    # Удаляем устаревшие файлы .csr и .cer (если есть)
    rm -f "$PROFILE_DIR"/r16.rosreestr.ru.csr
    rm -f "$PROFILE_DIR"/rosreestr.ru.cer

    cd "$PROFILE_DIR" || { echo "Не удалось перейти в $PROFILE_DIR" >&2; exit 1; }
    for name in "${CERT_NAMES[@]}"; do
        echo "Обработка сертификата $name"
        # Удаляем старую запись, если есть
        if run_as_cyber_root certutil -L -n "$name" -d sql:. >/dev/null 2>&1; then
            echo "  Удаляем существующую запись $name"
            run_as_cyber_root certutil -D -n "$name" -d sql:. || true
        fi
        # Добавляем новую (только .crt)
        if [ -f "${name}.crt" ]; then
            run_as_cyber_root certutil -A -n "$name" -t "C,C,C" -i "${name}.crt" -d sql:.
            echo "  Сертификат $name добавлен в NSS."
        else
            echo "  Файл ${name}.crt отсутствует, пропускаем."
        fi
    done
fi

# === 3. Работа с пользовательской базой NSS (Chromium) ===
echo "=== Обновление сертификатов в базе NSS пользователя cyber-root (Chromium) ==="
cd "$SOURCE_DIR" || { echo "Не удалось перейти в $SOURCE_DIR" >&2; exit 1; }

# Корневой сертификат
run_as_cyber_root certutil -d sql:/home/cyber-root/.pki/nssdb -D -n "Russian Trusted Root" 2>/dev/null || true
if [ -f "Корн Russian Trusted root.pem" ]; then
    run_as_cyber_root certutil -d sql:/home/cyber-root/.pki/nssdb -A -t "C,," -n "Russian Trusted Root" -i "Корн Russian Trusted root.pem"
    echo "  Корневой сертификат добавлен."
else
    echo "Предупреждение: файл 'Корн Russian Trusted root.pem' не найден, пропускаем импорт." >&2
fi

# Промежуточный сертификат
run_as_cyber_root certutil -d sql:/home/cyber-root/.pki/nssdb -D -n "Russian Trusted Sub" 2>/dev/null || true
if [ -f "Корн Russian Trusted sub.pem" ]; then
    run_as_cyber_root certutil -d sql:/home/cyber-root/.pki/nssdb -A -t "C,," -n "Russian Trusted Sub" -i "Корн Russian Trusted sub.pem"
    echo "  Промежуточный сертификат добавлен."
else
    echo "Предупреждение: файл 'Корн Russian Trusted sub.pem' не найден, пропускаем импорт." >&2
fi

# Сертификат r16.rosreestr.ru
run_as_cyber_root certutil -d sql:/home/cyber-root/.pki/nssdb -D -n "r16.rosreestr.ru" 2>/dev/null || true
if [ -f "r16.rosreestr.ru.crt" ]; then
    run_as_cyber_root certutil -d sql:/home/cyber-root/.pki/nssdb -A -t "P,," -n "r16.rosreestr.ru" -i r16.rosreestr.ru.crt
    echo "  Сертификат r16.rosreestr.ru добавлен."
else
    echo "Предупреждение: файл r16.rosreestr.ru.crt не найден, пропускаем импорт." >&2
fi

# Сертификат rosreestr.ru
run_as_cyber_root certutil -d sql:/home/cyber-root/.pki/nssdb -D -n "rosreestr.ru" 2>/dev/null || true
if [ -f "rosreestr.ru.crt" ]; then
    run_as_cyber_root certutil -d sql:/home/cyber-root/.pki/nssdb -A -t "P,," -n "rosreestr.ru" -i rosreestr.ru.crt
    echo "  Сертификат rosreestr.ru добавлен."
else
    echo "Предупреждение: файл rosreestr.ru.crt не найден, пропускаем импорт." >&2
fi

# === 4. Обновление системного хранилища CA ===
echo "=== Обновление системного хранилища CA ==="
# Копируем сертификаты в /usr/local/share/ca-certificates/
for name in "${CERT_NAMES[@]}"; do
    if [ -f "$SOURCE_DIR/${name}.crt" ]; then
        cp -v "$SOURCE_DIR/${name}.crt" /usr/local/share/ca-certificates/
    fi
done
update-ca-certificates -v --fresh

# === 5. Перезапуск служб (только nginx, если установлен) ===
echo "=== Перезапуск служб ==="
if command -v systemctl >/dev/null 2>&1; then
    if systemctl is-active --quiet nginx; then
        systemctl restart nginx && echo "nginx перезапущен." || echo "Не удалось перезапустить nginx" >&2
    fi
else
    echo "systemctl не найден, пропускаем перезапуск служб." >&2
fi

echo "=== Готово ==="

# Дополнительная информация о скопированных файлах
ls -la /ds-data/ssl/
ls -la /etc/nginx/ssl/
ls -la /var/www/r7-office/Data/ssl/
```
