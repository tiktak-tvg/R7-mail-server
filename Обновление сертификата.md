#!/bin/bash

set -euo pipefail

# === Конфигурация ===
SOURCE_DIR="/mnt/CDDiskPack/CDinstall_Astra_1.7.4/sslcert"
TARGET_DIRS=(
    "/var/www/r7-office/Data/ssl"
    "/ds-data/ssl"
    "/etc/nginx/ssl"
)
PROFILE_PATTERN="/home/cyber-root/.r7organizer/*.default-default"
CERT_NAMES=("r16.rosreestr.ru")   # имена без расширения

# === Проверка прав ===
if [ "$EUID" -ne 0 ]; then
    echo "Ошибка: скрипт должен быть запущен от root" >&2
    exit 1
fi

# === Проверка исходных файлов ===
missing=0
for name in "${CERT_NAMES[@]}"; do
    if [ ! -f "$SOURCE_DIR/${name}.crt" ]; then
        echo "Ошибка: исходный файл $SOURCE_DIR/${name}.crt не найден" >&2
        missing=1
    fi
    # Ключ может отсутствовать (некоторые сертификаты поставляются без ключа)
    if [ ! -f "$SOURCE_DIR/${name}.key" ]; then
        echo "Предупреждение: исходный ключ $SOURCE_DIR/${name}.key не найден" >&2
    fi
done
if [ $missing -eq 1 ]; then
    exit 1
fi

# === Функция выполнения от имени cyber-root ===
run_as_cyber_root() {
    sudo -u cyber-root "$@"
}

# === 1. Копирование файлов в целевые папки ===
echo "=== Копирование сертификатов в целевые каталоги ==="
for dir in "${TARGET_DIRS[@]}"; do
    # Создать папку, если её нет
    mkdir -p "$dir"
    echo "--> $dir"
    for name in "${CERT_NAMES[@]}"; do
        for ext in crt key; do
            src="$SOURCE_DIR/${name}.${ext}"
            if [ -f "$src" ]; then
                cp -v "$src" "$dir/"
                # Установка прав: для ключей - 600, для сертификатов - 644 (по умолчанию)
                if [ "$ext" = "key" ]; then
                    chmod 600 "$dir/${name}.${ext}"
                fi
            fi
        done
    done
done

# === 2. Работа с профилем .r7organizer (NSS) ===
PROFILE_DIR=$(ls -d $PROFILE_PATTERN 2>/dev/null | head -n1)
if [ -z "$PROFILE_DIR" ]; then
    echo "ОШИБКА: Каталог профиля не найден по шаблону $PROFILE_PATTERN" >&2
else
    echo "=== Обновление сертификатов в профиле NSS: $PROFILE_DIR ==="
    # Копируем свежие файлы в профиль (если их там ещё нет)
    for name in "${CERT_NAMES[@]}"; do
        for ext in crt key; do
            src="$SOURCE_DIR/${name}.${ext}"
            if [ -f "$src" ]; then
                cp -v "$src" "$PROFILE_DIR/"
            fi
        done
    done

    cd "$PROFILE_DIR" || { echo "Не удалось перейти в $PROFILE_DIR" >&2; exit 1; }
    for name in "${CERT_NAMES[@]}"; do
        echo "Обработка сертификата $name"
        # Удаляем старую запись, если есть
        if run_as_cyber_root certutil -L -n "$name" -d sql:. >/dev/null 2>&1; then
            echo "  Удаляем существующую запись $name"
            run_as_cyber_root certutil -D -n "$name" -d sql:.
        fi
        # Добавляем новую (только .crt)
        if [ -f "${name}.crt" ]; then
            # Флаги "C,C,C" подходят для корневого/промежуточного CA. Для сертификата сервера могут потребоваться другие.
            run_as_cyber_root certutil -A -n "$name" -t "C,C,C" -i "${name}.crt" -d sql:.
            echo "  Сертификат $name добавлен в NSS."
        else
            echo "  Файл ${name}.crt отсутствует, пропускаем."
        fi
    done
fi

# === 3. Обновление системного хранилища CA ===
echo "=== Обновление системного хранилища CA ==="
# Принудительно копируем сертификаты в /usr/local/share/ca-certificates/
# чтобы они попали в обновление (если там их ещё нет)
for name in "${CERT_NAMES[@]}"; do
    if [ -f "$SOURCE_DIR/${name}.crt" ]; then
        cp -v "$SOURCE_DIR/${name}.crt" /usr/local/share/ca-certificates/
    fi
done
update-ca-certificates -v --fresh

# === 4. Перезапуск служб ===
echo "=== Перезапуск служб ==="
if command -v systemctl >/dev/null 2>&1; then
    if systemctl is-active --quiet nginx; then
        systemctl restart nginx && echo "nginx перезапущен." || echo "Не удалось перезапустить nginx" >&2
    fi

    if systemctl list-units --full -all | grep -q "r7-office"; then
        systemctl restart r7-office-* 2>/dev/null && echo "Службы R7-Office перезапущены." || echo "Не удалось перезапустить R7-Office (проверьте вручную)." >&2
    fi
else
    echo "systemctl не найден, пропускаем перезапуск служб." >&2
fi

echo "=== Готово ==="

# === 5. Отдельная работа с профилем .r7organizer (NSS) ===
PROFILE_DIR=$(ls -d $PROFILE_PATTERN 2>/dev/null | head -n1)
if [ -z "$PROFILE_DIR" ]; then
    echo "ОШИБКА: Каталог профиля не найден по шаблону $PROFILE_PATTERN"
else
    echo "=== Обновление сертификатов в профиле NSS: $PROFILE_DIR ==="
    # Копируем свежие файлы в профиль (если их там ещё нет)
    for name in "${CERT_NAMES[@]}"; do
        for ext in crt key; do
            src="$NEW_CERTS_DIR/${name}.${ext}"
            if [ -f "$src" ]; then
                cp -v "$src" "$PROFILE_DIR/"
            fi
        done
    done

    # Переходим в профиль и обновляем базу данных NSS
    cd "$PROFILE_DIR" || exit 1
    for name in "${CERT_NAMES[@]}"; do
        echo "Обработка сертификата $name"
        # Удаляем старую запись (если есть)
        run_as_cyber_root certutil -D -n "$name" -d sql:. 2>/dev/null
        # Добавляем новую (только для .crt)
        if [ -f "${name}.crt" ]; then
            run_as_cyber_root certutil -A -n "$name" -t "C,C,C" -i "${name}.crt" -d sql:.
            echo "  Сертификат $name добавлен в NSS."
        else
            echo "  Файл ${name}.crt отсутствует, пропускаем."
        fi
    done
fi

# === 6. Обновление системного хранилища CA (генерирует .pem) ===
echo "=== Обновление системного хранилища CA ==="
# Проверяем, есть ли сертификаты в /usr/local/share/ca-certificates/ (могли быть заменены на шаге 1)
if ls /usr/local/share/ca-certificates/ | grep -q "\.crt$"; then
    update-ca-certificates -v --fresh
else
    echo "В /usr/local/share/ca-certificates/ нет .crt файлов. Обновление пропущено."
fi

# === 7. Перезапуск служб ===
echo "=== Перезапуск служб ==="
if systemctl is-active --quiet nginx; then
    systemctl restart nginx && echo "nginx перезапущен."
fi

if systemctl list-units --full -all | grep -q "r7-office"; then
    systemctl restart r7-office-* 2>/dev/null && echo "Службы R7-Office перезапущены." || echo "Не удалось перезапустить R7-Office (проверьте вручную)."

#!/bin/bash

# Скрипт для обновления сертификатов Росреестра (версия с r16)
# - удаляет старые сертификаты в локальном профиле
# - копирует новые файлы с удалённого сервера root@10.16.100.18
# - добавляет сертификаты в локальную базу NSS

# Конфигурация
REMOTE_USER="root"
REMOTE_HOST="10.16.100.18"
REMOTE_PROFILE_PATTERN="/home/cyber-root/.r7organizer/*.default-default"
LOCAL_PROFILE_PATTERN="/home/cyber-root/.r7organizer/*.default-default"

# Имена сертификатов (r16)
CERT_NAMES=("r16.rosreestr.ru" "rosreestr.ru")

# ----------------------------------------------------------------------
# Функция для выполнения команд от имени пользователя cyber-root (если запущено от root)
run_as_cyber_root() {
    if [ "$EUID" -eq 0 ]; then
        sudo -u cyber-root "$@"
    else
        "$@"
    fi
}

# ----------------------------------------------------------------------
# 1. Определяем локальный каталог профиля
LOCAL_PROFILE_DIR=$(ls -d $LOCAL_PROFILE_PATTERN 2>/dev/null | head -n1)
if [ -z "$LOCAL_PROFILE_DIR" ]; then
    echo "ОШИБКА: Локальный каталог профиля не найден по шаблону $LOCAL_PROFILE_PATTERN"
    exit 1
fi

# Переходим в локальный каталог (важно для certutil)
cd "$LOCAL_PROFILE_DIR" || { echo "Не удалось перейти в $LOCAL_PROFILE_DIR"; exit 1; }
echo "Локальный каталог профиля: $(pwd)"

# ----------------------------------------------------------------------
# 2. Определяем удалённый каталог профиля (раскрываем * через ssh)
echo "Определяем удалённый каталог профиля на $REMOTE_HOST..."
REMOTE_PROFILE_DIR=$(ssh ${REMOTE_USER}@${REMOTE_HOST} "ls -d ${REMOTE_PROFILE_PATTERN} 2>/dev/null | head -n1")
if [ -z "$REMOTE_PROFILE_DIR" ]; then
    echo "ОШИБКА: Не удалось найти каталог профиля на удалённом сервере по шаблону $REMOTE_PROFILE_PATTERN"
    exit 1
fi
echo "Удалённый каталог профиля: $REMOTE_PROFILE_DIR"

# ----------------------------------------------------------------------
# 3. Удаляем старые сертификаты из локальной базы
echo "Удаляем предыдущие версии сертификатов..."
for cert in "${CERT_NAMES[@]}"; do
    run_as_cyber_root certutil -D -n "$cert" -d sql:. 2>/dev/null \
        && echo "  - удалён $cert" || echo "  - сертификат $cert не найден (пропускаем)"
done

# ----------------------------------------------------------------------
# 4. Копируем файлы с удалённого сервера (используем явные имена)
echo "Копируем файлы сертификатов с удалённого сервера..."
for cert in "${CERT_NAMES[@]}"; do
    # Копируем все файлы, начинающиеся с этого имени (например, .crt, .key, и т.д.)
    scp "${REMOTE_USER}@${REMOTE_HOST}:${REMOTE_PROFILE_DIR}/${cert}.*" . 2>/dev/null
    if [ $? -eq 0 ]; then
        echo "  - скопированы файлы для $cert"
    else
        echo "  - ВНИМАНИЕ: не удалось скопировать файлы для $cert (возможно, отсутствуют на сервере)"
    fi
done

# ----------------------------------------------------------------------
# 5. Проверяем наличие CRT-файлов и добавляем сертификаты
echo "Добавляем новые сертификаты в локальную базу..."
for cert in "${CERT_NAMES[@]}"; do
    crt_file="${cert}.crt"
    if [ -f "$crt_file" ]; then
        run_as_cyber_root certutil -A -n "$cert" -t "C,C,C" -i "$crt_file" -d sql:.
        if [ $? -eq 0 ]; then
            echo "  + сертификат $cert успешно добавлен"
        else
            echo "  ! ОШИБКА при добавлении сертификата $cert"
        fi
    else
        echo "  ! Файл $crt_file не найден, сертификат $cert не добавлен"
    fi
done
echo "Готово." 
fi
echo "=== Готово ==="
